@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun MainScreen() {
//    State untuk menyimpan teks Bercode yang diketik
    val  context = LocalContext.current

    var barcodeInput by remember { mutableStateOf("") }

    var showScanner by remember { mutableStateOf(false) }

    val productList = listOf(
        Product("12345", "Indomie Goreng", 3500), // Ganti dengan barcode asli barangmu!
        Product("123456", "Kopi Kapal Api", 10000),
        Product("6950376733412", "Kopi Kapal Api", 5000)
    )

    var cartItems by remember { mutableStateOf(listOf<Product>()) }

//   State untuk PopUp Pembayaran
    var showPaymentDialog by remember { mutableStateOf(false) }
    var cashInput by remember { mutableStateOf("") }

//   State untuk PopUp Sukses (Kembalian)
    var showSuccessDialog by remember { mutableStateOf(false) }
    var changeAmount by remember { mutableStateOf(0) }

    val totalHarga = cartItems.sumOf { it.price * it.quantity }

    val cameraPermissionLauncher = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.RequestPermission()
    ) { isGranted: Boolean ->
        if (isGranted) {
            showScanner = true
        } else {
            println("Izin Kamera Ditolak")
        }
    }

    Scaffold(
        topBar = {
//            Ini adalah Header
            TopAppBar(
                title = { Text("Toko Kasir MiniMarket", color = Color.White) },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = Color(0xFFFF8C00)
                )
            )
        },
        bottomBar = {
            Surface(shadowElevation = 8.dp,
                modifier =
                    Modifier.navigationBarsPadding()) {
                Column(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(16.dp)
                ) {
                    // 1. Input Barcode & Tombol Scan
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        OutlinedTextField(
                            value = barcodeInput,
                            onValueChange = { barcodeInput = it },
                            label = { Text("Masukkan Barcode") },
                            modifier = Modifier.weight(1f),
                            singleLine = true,

                            leadingIcon = {
                                Icon(
                                  imageVector = Icons.Default.ViewWeek,
                                  contentDescription = "Barcode Icon",
                                  tint = Color.Gray
                                )
                            },
//                            Taruh tombol scan di dalam kolom (ujuang kanan)
                            trailingIcon = {
                                IconButton(onClick = {
//                                    Logika buka kamera tetap sama
                                    val permissionCheckResult = ContextCompat.checkSelfPermission(context, android.Manifest.permission.CAMERA)
                                    if (permissionCheckResult == android.content.pm.PackageManager.PERMISSION_GRANTED) {
                                        showScanner = true
                                    } else {
                                        cameraPermissionLauncher.launch(android.Manifest.permission.CAMERA)
                                    }
                                }) {
//                                    Kamu bisa pakai Icon Kamera agar lebih efisien
                                    Icon(
                                        imageVector = Icons.Default.PhotoCamera,
                                        contentDescription = "Scan",
                                        tint = Color(0xFFFF8C00)
                                    )
                                }
                            }
                        )
                        Spacer(modifier = Modifier.width(8.dp))
                        // Tombol ini sekarang fungsinya untuk INPUT MANUAL (Tambah ke keranjang)
                        Button(
                            onClick = {
                                val foundProduct = productList.find { it.barcode == barcodeInput }
                                if (foundProduct != null) {
                                    val existingItem = cartItems.find { it.barcode == foundProduct.barcode }
                                    if (existingItem != null) {
                                        // Jika sudah ada, tambah jumlahnya saja
                                        cartItems = cartItems.map {
                                            if (it.barcode == foundProduct.barcode) it.copy(quantity = it.quantity + 1) else it
                                        }
                                    } else {
// Jika belum ada, tambah baris baru
                                        cartItems = cartItems + foundProduct.copy(timestampId = System.currentTimeMillis())                                    }
                                    barcodeInput = "" // Bersihkan input setelah tambah
                                }
                            },
                            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFF8C00)),
                            shape = RoundedCornerShape(8.dp),
                            modifier = Modifier.height(56.dp) // Samakan tingginya dengan TextField
                        ) {
                            Text("TAMBAH")
                        }
                    }

                    Spacer(modifier = Modifier.height(16.dp))

                    // 2. Total Harga & Tombol Bayar
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.SpaceBetween, // Diganti agar pas
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Column {

                            Text("Total Bayar", fontSize = 14.sp)
                            // Jangan lupa import FontWeight dari androidx.compose.ui.text.font
                            AnimatedContent(targetState = totalHarga) { targetTotal ->
                                Text(
                                    text = "Rp $targetTotal",
                                    fontSize = 24.sp,
                                    color = Color(0xFFFF8C00),
                                    fontWeight = FontWeight.Bold
                                )
                            }
                        }

                        Button(
                            onClick = { /* Simpan Transaksi */
                            if (cartItems.isNotEmpty()){
                                showPaymentDialog = true // Muncul PopUp jika keranjang tidak kosong.
                                }
                            },
                            modifier = Modifier.height(50.dp),
                            colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFF8C00))
                        ) {
                            // Kamu butuh import Icons & ShoppingCart
                            Icon(Icons.Default.ShoppingCart, contentDescription = null )
                            Spacer(Modifier.width(8.dp))
                            Text("BAYAR")
                        }
                    }
                }
            }
        },
    ) { innerPadding ->
// Bagian tengah (Daftar Belanja)
        Box(modifier = Modifier.padding(innerPadding).fillMaxSize()) {
            if ( cartItems.isEmpty()) {
            Text("Belum ada barang. Silahkan SCAN!", modifier = Modifier.align(Alignment.Center))
            } else {
                LazyColumn(
                    modifier = Modifier.fillMaxSize(),
                    contentPadding = PaddingValues(16.dp),
                    verticalArrangement = Arrangement.spacedBy(8.dp) // Jarak antar baris barang
                ) {
                    items(
                        items = cartItems,
                        key = { it.timestampId }
                    ) { produk ->
                        var isDismissed by remember { mutableStateOf(false) }
                        val dismissState = rememberSwipeToDismissBoxState(
                            confirmValueChange = { value ->
                                // SEKARANG: Geser dari Kiri ke Kanan (StartToEnd)
                                if (value == SwipeToDismissBoxValue.StartToEnd) {
                                    isDismissed = true // Aktifkan animasi keluar
                                    true
                                } else {
                                    false
                                }
                            }
                        )

                        // 2. LaunchedEffect: Tunggu animasi selesai (300ms) baru hapus data aslinya
                        LaunchedEffect(isDismissed) {
                            if (isDismissed) {
                                kotlinx.coroutines.delay(300) // Waktu yang sama dengan durasi animasi
                                cartItems = cartItems.filter { it != produk }
                            }
                        }

                        AnimatedVisibility(
                            visible = cartItems.contains(produk),
                            exit = fadeOut(animationSpec = tween(300)) + shrinkVertically()
                        ) {
                            SwipeToDismissBox(
                                state = dismissState,
                                backgroundContent = {
                                    // Warna merah muncul saat geser ke kanan
                                    val color = if (dismissState.dismissDirection == SwipeToDismissBoxValue.StartToEnd)
                                        Color.Red else Color.Transparent

                                    Box(
                                        modifier = Modifier
                                            .fillMaxSize()
                                            .background(color, shape = RoundedCornerShape(8.dp))
                                            .padding(horizontal = 20.dp),
                                        contentAlignment = Alignment.CenterStart // Ikon sampah di KIRI
                                    ) {
                                        Icon(
                                            imageVector = Icons.Default.Delete,
                                            contentDescription = "Hapus",
                                            tint = Color.White
                                        )
                                    }
                                },
                                // AKTIFKAN geser ke kanan, matikan geser ke kiri
                                enableDismissFromStartToEnd = true,
                                enableDismissFromEndToStart = false
                            ) {
//                                Tampilan Produk
                                Card(
                                    modifier = Modifier.fillMaxWidth(1f),
                                    shape = RoundedCornerShape(8.dp)
                                ) {
                                    Row(
                                        modifier = Modifier
                                            .fillMaxWidth()
                                            .padding(16.dp),
                                        horizontalArrangement = Arrangement.SpaceBetween,
                                        verticalAlignment = Alignment.Top // Supaya teks dan tombol sejajar tinggi tengahnya
                                    ) {
                                        // Bagian Kiri: Info Produk
                                        Column(modifier = Modifier.weight(1f)) { // Gunakan weight supaya teks tidak menabrak tombol
                                            Text(produk.name, fontWeight = FontWeight.Bold, maxLines = 1)
                                            Text("Barcode: ${produk.barcode}", fontSize = 12.sp, color = Color.Gray)
                                            Text("Rp ${produk.price}", color = Color(0xFFFF8C00), fontWeight = FontWeight.SemiBold)
                                        }
                                        // Bagian Kanan: Quantity & Tombol
                                        Column(
                                            horizontalAlignment = Alignment.CenterHorizontally
                                        ) {
//                                            Angka Quantity
                                            Text(
                                                text = "${produk.quantity}",
                                                fontWeight = FontWeight.Bold,
                                                fontSize = 22.sp,
                                                modifier = Modifier.padding(bottom = 2.dp) //Jarak tombol di bawah
                                            )
//                                          Tombol dengan Jarak yang lebar
                                            Row (
                                                verticalAlignment = Alignment.CenterVertically,
                                                horizontalArrangement = Arrangement.spacedBy(12.dp) // Jarak Antara Tombol
                                            ) {
                                                // Tombol Kurang [-]
                                                IconButton(
                                                    onClick = {
                                                        if (produk.quantity > 1) {
                                                            cartItems = cartItems.map {
                                                                if (it.barcode == produk.barcode) it.copy(quantity = it.quantity - 1) else it
                                                            }
                                                        }
                                                    },
                                                    modifier = Modifier.size(48.dp)
                                                ) {
                                                    Icon(Icons.Default.Remove, contentDescription = null, tint = Color.Gray, modifier = Modifier.size(24.dp))
                                                }

                                                // Tombol Tambah [+]
                                                IconButton(
                                                    onClick = {
                                                        cartItems = cartItems.map {
                                                            if (it.barcode == produk.barcode) it.copy(quantity = it.quantity + 1) else it
                                                        }
                                                    },
                                                    modifier = Modifier.size(48.dp)
                                                ) {
                                                    Icon(Icons.Default.Add, contentDescription = null, tint = Color(0xFFFF8C00), modifier = Modifier.size(24.dp))
                                                }
                                            }

                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

//        LOGIKA POP-UP PEMBAYARAN
        if (showPaymentDialog) {
            val cash = cashInput.toIntOrNull() ?: 0
            val isUangKurang = cash < totalHarga && cashInput.isNotEmpty()

            AlertDialog(
                onDismissRequest = { showPaymentDialog = false },
                title = { Text("Proses Pembayaran", fontWeight = FontWeight.Bold)},
                text = {
                    Column{



                        Text("Total Belanja: Rp. $totalHarga", fontSize = 18.sp,
                            color = if (isUangKurang) Color.Red else Color(0xFFFF8C00), fontWeight = FontWeight.Bold
                        )
                        Spacer(modifier = Modifier.height(16.dp))
                        OutlinedTextField(
                            value = cashInput,
                            onValueChange = {cashInput = it},
                            label = { Text("Uang Tunai (Rp)")},
                            singleLine = true,
                            isError = isUangKurang,
                            keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
                                keyboardType = androidx.compose.ui.text.input.KeyboardType.Number
                        ),
//                        Menambahkan teks peringatan kecil di bawah input
                            supportingText = {
                                if (isUangKurang) {
                                    Text ("Uang Tidak Cukup", color = Color.Red)
                                }
                            }
                        )
                    }
                },
                confirmButton = {
                    Button(
                        onClick = {

                            if (cash >= totalHarga) {
                                changeAmount = cash - totalHarga
                                showPaymentDialog = false // Tutup PopUp Bayar
                                showSuccessDialog = true //Buka PopUp Sukses

                                 }
                            },
                        // 4. Tombol jadi abu-abu kalau uang belum cukup (disable secara visual)
                        colors = ButtonDefaults.buttonColors(
                            containerColor = if (cash >= totalHarga) Color(0xFFFF8C00) else Color.Gray
                        ),
                        enabled = cash >= totalHarga // Tombol tidak bisa diklik kalau kurang
                    ) {
                        Text("Proses")
                    }
                },
                dismissButton = {
                    TextButton(onClick = {showPaymentDialog = false}) {
                        Text("Batal", color = Color.Gray)
                    }
                }
            )
        }

//      LOGIKA POP-UP SUKSES & KEMBALIAN
        if (showSuccessDialog) {
            AlertDialog(
                onDismissRequest = {
//                    Jika di klik di luar, maka dialog hilang semua
                    showSuccessDialog = false
                    cartItems = emptyList()
                    cashInput = ""
                },
                title = { Text("Transaksi Berhasil", color = Color(0xFFFF8C00), fontWeight = FontWeight.Bold)},
                text = {
                    Column {
                        Text("Kembalian Pelanggan:", fontSize = 16.sp)
                        Text("Rp $changeAmount", fontSize = 24.sp, fontWeight = FontWeight.Bold)
                    }
                },
                confirmButton = {
                    Button(
                        onClick = {
//                            Reset Keranjang Ketika Selesai
                            showSuccessDialog = false
                            cartItems = emptyList() //Mengosongkan Keranjang
                            cashInput = "" //Mengosongkan Input Uang
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFFF8C00))
                    ) {
                        Text("Selesai & Kasir Baru")
                    }
                }
            )
        }


//      Logika POP-UP Kamera
        if (showScanner){
            AlertDialog(
                onDismissRequest = {showScanner = false},
               confirmButton = {
                TextButton(
                   onClick = { showScanner = false },
                    modifier = Modifier.padding(top = 0.dp)

               ) {
                   Text("TUTUP", fontWeight = FontWeight.Bold)
               }
        },
                title = {Text("Scan Barcode Produk")},
                text = {
                    Box(modifier = Modifier.size(300.dp)
                        .clipToBounds()) {
//                        Memanggil file BarcodeScanner.Kt yang kamu buat tadi!
                        BarcodeScannerView(
                            onBarcodeDetected = {code ->
//                                LOGIKA: Cari di list, apakah barcode ini ada di gudang?
                                val foundProduct = productList.find { it.barcode == code }
                                val toneGen = ToneGenerator(AudioManager.STREAM_MUSIC, 100)
                                toneGen.startTone(ToneGenerator.TONE_PROP_BEEP, 150)

                                if (foundProduct != null) {
                                    val existingItem = cartItems.find { it.barcode == foundProduct.barcode }
                                    if (existingItem != null) {
                                        // Jika sudah ada, update jumlahnya
                                        cartItems = cartItems.map {
                                            if (it.barcode == foundProduct.barcode) it.copy(quantity = it.quantity + 1) else it
                                        }
                                    } else {
                                        // Jika barang baru, tambahkan ke list
                                        cartItems = cartItems + foundProduct.copy(timestampId = System.currentTimeMillis())
                                    }
                                } else {
// Jika barcode tidak terdaftar di productList
                                    barcodeInput = code
                                }
                                showScanner = false // Tutup kamera setelah berhasil
                            },
                            modifier = Modifier.fillMaxSize()
                        )
                    }
                }
            )
        }
    }
}

        }
    }
